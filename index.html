Installation rapide

# Linux / macOS
curl -Lo paserial https://github.com/jiiro974/paserial-release/releases/download/v0.3.7/paserial-$(uname -s | tr '[:upper:]' '[:lower:]')-$(uname -m | sed s/x86_64/amd64/)
chmod +x paserial && sudo mv paserial /usr/local/bin/

# Mise à jour
paserial self-update

Usage

paserial --port /dev/ttyUSB0 upgrade-10.2.8.txt
paserial --port /dev/ttyUSB0 --dry-run upgrade-10.2.8.txt
paserial debug-panorama --port /dev/ttyUSB0
